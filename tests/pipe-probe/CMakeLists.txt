cmake_minimum_required(VERSION 3.25)
project(MsimePipeProbe LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "The pipe probe exercises a real Windows Server process")
endif()
set(MSIME_CONTRACTS_DIR "" CACHE PATH "Contracts from the repository's engine submodule")
add_executable(msime-ipc-probe main.cpp)
target_compile_features(msime-ipc-probe PRIVATE cxx_std_17)
target_compile_definitions(msime-ipc-probe PRIVATE NOMINMAX UNICODE _UNICODE)
target_include_directories(msime-ipc-probe PRIVATE "${MSIME_CONTRACTS_DIR}")
if(MSVC)
    target_compile_options(msime-ipc-probe PRIVATE /utf-8)
endif()

add_executable(msime-translation-buffer-probe translation_buffer.cpp)
target_compile_features(msime-translation-buffer-probe PRIVATE cxx_std_17)
target_compile_definitions(msime-translation-buffer-probe PRIVATE NOMINMAX UNICODE _UNICODE)
target_include_directories(msime-translation-buffer-probe PRIVATE "${MSIME_CONTRACTS_DIR}")
target_link_libraries(msime-translation-buffer-probe PRIVATE winhttp)
if(MSVC)
    target_compile_options(msime-translation-buffer-probe PRIVATE /utf-8)
endif()
